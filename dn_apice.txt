drop database if exists dn_apice;
create database dn_apice;
use dn_apice;
create table bioma(nome varchar(100) not null primary key,
   estacao varchar(100));

create table organismo(nome varchar(100) not null primary key,
   idade int not null,
   bioma varchar(100) not null,
   hidratacao decimal(10, 2) not null,
   temperatura decimal(10, 2) not null,
   batimento decimal(10, 2) not null
);

create table coracao(batimento decimal(10, 2));

create table cerebro(
	oxigenacao int not null primary key,
   	impulsos_nervosos int not null
);
